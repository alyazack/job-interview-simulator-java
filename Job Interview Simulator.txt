    static Random random = new Random(); 
    
    static class Question { 
        String text; 
        List<String> keywords; 

        Question(String text, List<String> keywords) { 
            this.text = text; 
            this.keywords = keywords; 
        } 
    } 

    static class InterviewResponse { 
        String question; 
        String answer; 
        int score; 
        
        InterviewResponse(String question, String answer, int score) { 
            this.question = question; 
            this.answer = answer; 
            this.score = score; 
        } 
    } 
    static Map<String, List<Question>> occupationQuestions = new HashMap<>(); 
    public static void main(String[] args) { 
        initializeQuestions(); 
        String[] occupationArray = occupationQuestions.keySet().toArray(new String[0]); 
        String selectedOccupation = (String) JOptionPane.showInputDialog(
        		null, 
                "Choose an occupation:", 
                "Occupation Selection", 
                JOptionPane.PLAIN_MESSAGE, 
                null, 

                occupationArray, 

                occupationArray[0]); 

        if (selectedOccupation == null) { 
            JOptionPane.showMessageDialog(null, "No occupation selected. Exiting."); 
            return; 
        } 

        List<Question> questions = occupationQuestions.get(selectedOccupation); 
        Collections.shuffle(questions); 

        List<InterviewResponse> responses = new ArrayList<>();

        for (int i = 0; i < 5; i++) { 
            Question q = questions.get(i); 
            String answer = JOptionPane.showInputDialog(null, "Q" + (i + 1) + ": " + q.text); 

            if (answer == null) { 
                answer = "(No answer)"; 
            } 

            int score = evaluateAnswer(answer, q.keywords); 
            responses.add(new InterviewResponse(q.text, answer, score)); 

        } 
        showResults(responses, selectedOccupation); 
    } 

    static int evaluateAnswer(String answer, List<String> keywords) { 
        int score = random.nextInt(31) + 50; // 50–80 
        for (String keyword : keywords) { 
            if (answer.toLowerCase().contains(keyword.toLowerCase())) { 
                score += 4; 
            } 
        } 
        return Math.min(score, 100); 
    } 

    static void showResults(List<InterviewResponse> responses, String occupation) { 
        StringBuilder result = new StringBuilder(); 
        int totalScore = 0; 
        result.append("Interview Summary for ").append(occupation).append(":\n\n"); 

        for (int i = 0; i < responses.size(); i++) { 
            InterviewResponse r = responses.get(i); 
            result.append("Q").append(i + 1).append(": ").append(r.question).append("\n"); 
            result.append("Your Answer: ").append(r.answer).append("\n"); 
            result.append("Score: ").append(r.score).append("/100\n\n"); 
            totalScore += r.score; 
        } 

        double average = totalScore / 5.0; 
        result.append(String.format("Average Score: %.2f/100\n", average)); 
        if (average >= 85) { 
            result.append("🎉 Excellent! You’re a great fit for the job."); 
        } else if (average >= 70) { 
            result.append("🙂 Good job. You may need a bit more preparation."); 
        } else { 
            result.append("😬 Needs improvement. Consider brushing up before the next interview."); 
        } 

        JOptionPane.showMessageDialog(null, result.toString(), "Interview Result", JOptionPane.INFORMATION_MESSAGE); 
    }
